from ..helpers.triage import (

class SolicitationNotFoundError(LookupError):
    pass


class SolicitationFinishedError(ValueError):
    pass


class SolicitationService:
    def __init__(self, repository: SolicitationRepository) -> None:
        self.repository = repository

    async def persist_from_analysis(
        self,
        payload: SolicitationCreate,
        *,
        solicitation_id: UUID | None = None,
    ) -> Solicitation:
        analysis = payload.analysis
        chat = await self.repository.create_session(
            build_triage_session_title(payload.input, analysis)
        )
        await self.repository.add_message(
            chat.id,
            payload.input.message_text,
            role=ChatMessageRole.USER,
        )
        await self.repository.add_message(
            chat.id,
            format_triage_analysis(analysis),
            role=ChatMessageRole.AGENT,
        )

        solicitation = Solicitation(
            id=solicitation_id or uuid4(),
            status=SolicitationStatus.RECEIVED,
            summary=analysis.summary,
            request_type=SolicitationType(analysis.request_type.value),
            urgency=SolicitationUrgency(analysis.urgency.value),
            session_id=chat.id,
        )
        self.repository.add(solicitation)
        await self.repository.session.flush()
        return solicitation

    async def create_from_analysis(
        self,
        payload: SolicitationCreate,
        *,
        solicitation_id: UUID | None = None,
    ) -> Solicitation:
        try:
            solicitation = await self.persist_from_analysis(
                payload,
                solicitation_id=solicitation_id or payload.id,
            )
        except Exception:
            await self._rollback()
            raise

        await self._commit()
        return solicitation

    async def list_solicitations(self) -> list[Solicitation]:
        return await self.repository.list_all()

    async def get_solicitation(self, solicitation_id: UUID) -> Solicitation:
        return await self._get_solicitation(solicitation_id)

    async def open_session(
        self,
        solicitation_id: UUID,
    ) -> tuple[Solicitation, ChatSession, list[ChatMessage], bool]:
        solicitation = await self._get_solicitation(
            solicitation_id,
            lock=True,
        )

        read_only = solicitation.status == SolicitationStatus.FINISHED
        if solicitation.status == SolicitationStatus.RECEIVED:
            solicitation.status = SolicitationStatus.IN_REVIEW

        await self._commit()

        chat = await self.repository.get_session(solicitation.session_id)
        if chat is None:
            raise SolicitationNotFoundError("SessÃ£o nÃ£o encontrada.")
        messages = await self.repository.list_messages(solicitation.session_id)
        return solicitation, chat, messages, read_only

    async def add_message(
        self,
        solicitation_id: UUID,
        content: str,
    ) -> ChatMessage:
        solicitation = await self._get_solicitation(
            solicitation_id,
            lock=True,
        )
        if solicitation.status == SolicitationStatus.FINISHED:
            raise SolicitationFinishedError(
                "A conversa estÃ¡ finalizada e nÃ£o aceita novas mensagens."
            )

        if solicitation.status == SolicitationStatus.RECEIVED:
            solicitation.status = SolicitationStatus.IN_REVIEW

        try:
            message = await self.repository.add_message(
                solicitation.session_id,
                content,
            )
        except Exception:
            await self._rollback()
            raise

        await self._commit()
        return message

    async def finalize(self, solicitation_id: UUID) -> Solicitation:
        solicitation = await self._get_solicitation(
            solicitation_id,
            lock=True,
        )

        solicitation.status = SolicitationStatus.FINISHED
        await self._commit()
        return solicitation

    async def _get_solicitation(
        self,
        solicitation_id: UUID,
        *,
        lock: bool = False,
    ) -> Solicitation:
        solicitation = await self.repository.get_by_id(
            solicitation_id,
            lock=lock,
        )
        if solicitation is None:
            raise SolicitationNotFoundError("SolicitaÃ§Ã£o nÃ£o encontrada.")
        return solicitation

    async def _commit(self) -> None:
        try:
            await self.repository.session.commit()
        except Exception:
            await self._rollback()
            raise

    async def _rollback(self) -> None:
        await self.repository.session.rollback()

class SolicitationRepository:
    def __init__(self, session: AsyncSession) -> None:
        self.session = session
        self.chat_repository = ChatRepository(session)

    def add(self, solicitation: Solicitation) -> None:
        self.session.add(solicitation)

    async def create_session(self, title: str) -> ChatSession:
        chat = ChatSession(user_id=None, title=title)
        self.session.add(chat)
        await self.session.flush()
        return chat

    async def get_session(self, session_id: UUID) -> ChatSession | None:
        return await self.session.get(ChatSession, session_id)

    async def list_all(self) -> list[Solicitation]:
        statement = (
            select(Solicitation)
            .join(ChatSession, Solicitation.session_id == ChatSession.id)
            .order_by(ChatSession.created_at.desc(), Solicitation.id.desc())
        )
        return list((await self.session.scalars(statement)).all())

    async def get_by_id(
        self,
        solicitation_id: UUID,
        *,
        lock: bool = False,
    ) -> Solicitation | None:
        statement = select(Solicitation).where(Solicitation.id == solicitation_id)
        if lock:
            statement = statement.with_for_update()
        return await self.session.scalar(statement)

    async def add_message(
        self,
        session_id: UUID,
        content: str,
        *,
        role: ChatMessageRole = ChatMessageRole.USER,
    ) -> ChatMessage:
        return await self.chat_repository.add_message(
            session_id,
            content,
            role=role,
        )

    async def list_messages(self, session_id: UUID) -> list[ChatMessage]:
        return await self.chat_repository.list_messages(session_id)

class ChatRepository:
    def __init__(self, session: AsyncSession) -> None:
        self.session = session

    async def create_session(self, user_id: UUID) -> ChatSession:
        chat = ChatSession(user_id=user_id)
        self.session.add(chat)
        await self.session.flush()
        return chat

    async def list_sessions(self, user_id: UUID) -> list[ChatSession]:
        stmt = (
            select(ChatSession)
            .where(ChatSession.user_id == user_id)
            .order_by(ChatSession.created_at.desc())
        )
        return list((await self.session.scalars(stmt)).all())

    async def get_session(self, session_id: UUID, user_id: UUID) -> ChatSession | None:
        stmt = select(ChatSession).where(
            ChatSession.id == session_id, ChatSession.user_id == user_id
        )
        return await self.session.scalar(stmt)

    async def add_message(
        self,
        session_id: UUID,
        content: str,
        *,
        role: ChatMessageRole = ChatMessageRole.USER,
    ) -> ChatMessage:
        next_position = await self.session.scalar(
            select(func.coalesce(func.max(ChatMessage.position), 0) + 1)
            .where(ChatMessage.session_id == session_id)
        )
        msg = ChatMessage(
            session_id=session_id,
            content=content,
            role=role,
            position=next_position,
            citations=[],
        )
        self.session.add(msg)
        await self.session.flush()
        return msg


    async def add_citation(self, message_id: UUID, chunk_id: UUID) -> Citation:
        citation = Citation(message_id=message_id, chunk_id=chunk_id)
        self.session.add(citation)
        await self.session.flush()
        return citation

    async def list_messages(self, session_id: UUID) -> list[ChatMessage]:
        stmt = (
            select(ChatMessage)
            .where(ChatMessage.session_id == session_id)
            .options(
                selectinload(ChatMessage.citations)
                .selectinload(Citation.chunk)
                .selectinload(DocumentChunk.document)
            )
            .order_by(
                ChatMessage.position.asc(),
                ChatMessage.created_at.asc(),
                ChatMessage.id.asc(),
            )
        )
        return list((await self.session.scalars(stmt)).all())

async def get_current_user(
    token: Annotated[str, Depends(oauth2_scheme)],
) -> User:
    session_maker = database.get_session_maker()

    try:
        async with session_maker() as session:
            user = await _build_auth_service(session).get_current_user(token)
    except InvalidAccessTokenError as error:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials.",
            headers={"WWW-Authenticate": "Bearer"},
        ) from error

    if user is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials.",
            headers={"WWW-Authenticate": "Bearer"},
        )

    return user

DEFAULT_DOCUMENT_UPLOAD_MAX_BYTES = 10 * 1024**2
MAX_DOCUMENT_UPLOAD_BYTES = 30 * 1024**2


def get_document_upload_max_bytes() -> int:
    max_bytes = globals()["MAX_DOCUMENT_UPLOAD_BYTES"]

    configured = os.getenv("DOCUMENT_UPLOAD_MAX_BYTES")
    if configured is None:
        return DEFAULT_DOCUMENT_UPLOAD_MAX_BYTES

    error_message = (
        f"DOCUMENT_UPLOAD_MAX_BYTES must be between 1 and {max_bytes}."
    )
    try:
        limit = int(configured)
    except ValueError as error:
        raise RuntimeError(error_message) from error

    if not 1 <= limit <= max_bytes:
        raise RuntimeError(error_message)

    return limit


def get_database_url() -> str:
    url = os.getenv("DATABASE_URL")
    if not url or not url.strip():
        raise RuntimeError("DATABASE_URL must be set.")
    return url.strip()

    return [
        origin.strip()
        for origin in configured_origins.split(",")
        if origin.strip()
    ]

    backend = os.getenv("DOCUMENT_STORAGE_BACKEND", "local").strip()
    if backend == "local" or backend == "gcs":

    raise RuntimeError("DOCUMENT_STORAGE_BACKEND must be 'local' or 'gcs'.")

    if not directory:
        raise RuntimeError("DOCUMENT_STORAGE_DIR must be set.")
    return Path(directory)
